(rule sovereignty/collapse
  :material-basis "a sovereign that cannot reproduce its claims sheds them"
  :fuel 4096
  (bindings
    (binding pressure :field sovereign/pressure))
  (when (> pressure 0.9i))
  (effects
    (add-node NodeType/SOCIAL_CLASS n1 (social-class/wealth 5$))
    (remove-node n1)
    (add-edge EdgeType/SOLIDARITY a b :strength 0.5c (solidarity/tension 0.1i))
    (remove-edge EdgeType/SOLIDARITY a b)
    (update-hyperedge h community/size (add 1))
    (update-node self sovereign/pressure (sub 0.1i))
    (remove-hyperedge h)))

(rule struggle/spark-mechanic
  :role mechanic
  :evidence designed
  :material-basis "repression supplies bounded pressure for a finite material transition"
  :fuel 384
  (bindings
    (binding repression-faced :field social-class/repression-faced)
    (binding spark-scale :const struggle/spark-scale)
    (binding spark-mass :expr
      (quantize-mass (* repression-faced spark-scale)))
    (binding no-incident-mass :expr (- 1m spark-mass)))
  (effects
    (choose :sample struggle/spark :slot 0
      (branch StruggleSparkOutcome/EXCESSIVE_FORCE
        :mass spark-mass
        (effects
          (update-node self social-class/last-incident-known (set 1))))
      (branch StruggleSparkOutcome/NO_INCIDENT
        :mass no-incident-mass
        (effects)))))
